(function(){
  const STORAGE_KEY = 'swiss_souvenir_list';
  const BOUGHT_KEY  = 'swiss_souvenir_bought';
  let items = {};   // id -> {flag, name, price, cat}

  const CAT_LABELS = {

  function load() {
    try { items  = JSON.parse(localStorage.getItem(STORAGE_KEY)) || {}; } catch(e) { items = {}; }

  function save() {
    try { localStorage.setItem(STORAGE_KEY, JSON.stringify(items)); } catch(e) {}
    try { localStorage.setItem(BOUGHT_KEY, JSON.stringify(bought)); } catch(e) {}
  }

  function updateButtons() {
    document.querySelectorAll('.ssl-btn').forEach(btn => {
      const id = btn.dataset.id;
      if (items[id]) {
        btn.classList.add('added');
        btn.textContent = '✓ 已加入清單';

        btn.classList.remove('added');
        btn.textContent = '+ 加入清單';
      }
    });
  }

  function updateCount() {
    document.getElementById('ssl-fab-count').textContent = Object.keys(items).length;
  }

  function groupedRows(showRemove) {
    const cats = ['supermarket', 'souvenir', 'store'];
    let html = '';
    cats.forEach(cat => {
      const ids = Object.keys(items).filter(id => items[id].cat === cat);
      if (!ids.length) return;
      html += `<div class="ssl-group">
        <div class="ssl-group-title">${CAT_LABELS[cat]}</div>`;
      ids.forEach(id => {
        const it = items[id];
        const isBought = !!bought[id];
        html += `<div class="ssl-list-row${isBought ? ' bought' : ''}">
          <div class="ssl-check${isBought ? ' checked' : ''}" onclick="sslToggleBought('${id}')">${isBought ? '✓' : ''}</div>
          <span class="ssl-list-flag">${it.flag}</span>
          <span class="ssl-list-name">${it.name}</span>
          ${it.img ? `<button class="ssl-list-img" onclick="sslShowImage('${it.img}','${it.name.replace(/'/g, "\\'")}')">看圖</button>` : ''}
          <span class="ssl-list-price">${it.price}</span>
          ${showRemove ? `<button class="ssl-list-remove" onclick="sslRemove('${id}')">×</button>` : ''}
        </div>`;
      });
      html += `</div>`;
    });
    return html;
  }

  function renderLists() {
    const ids = Object.keys(items);
    const modalList = document.getElementById('ssl-modal-list');
    const finalList = document.getElementById('ssl-final-list');

    if (!ids.length) {
      modalList.innerHTML = '<div class="ssl-empty">還沒有加入任何商品<br>滑動文章並點選「加入清單」吧！</div>';
      finalList.innerHTML = '<div class="ssl-empty">還沒有加入任何商品，往上滑動選擇你想買的吧！</div>';
      return;
    }

    modalList.innerHTML = groupedRows(true);
    finalList.innerHTML = groupedRows(false);
  }

  function refresh() {
    updateButtons();
    updateCount();
    renderLists();
  }

  document.querySelectorAll('.ssl-btn').forEach(btn => {

      if (items[id]) {

        delete bought[id];
      } else {
        items[id] = { flag: btn.dataset.flag, name: btn.dataset.name, price: btn.dataset.price, cat: btn.dataset.cat, img: btn.dataset.img || '' };
      }
      save();
      refresh();
    });
  });

    delete items[id];
    delete bought[id];
    save();
    refresh();

  window.sslToggleBought = function(id) {
    bought[id] = !bought[id];
    save();

  window.sslShowImage = function(src, alt) {

  window.sslCloseImage = function() {
    document.getElementById('ssl-lightbox').classList.remove('show');
  };
  window.sslClearAll = function() {
    items = {};
    bought = {};
    save();
    refresh();
  };
  window.sslCopyList = function() {

    if (!ids.length) return;

        lines.push(`${mark} ${it.flag} ${it.name}（${it.price}）`);

## 結論

以上是我旅瑞多年來覺得可買以及過譽的物品，順便一提，超市 Coop 的自有品牌是 Prix，Migros 自有品牌是 M Budget，就是一些便宜堪用的東西。想要買回去打發同事或同學的話，也可以買這兩個牌子的巧克力，價差大概是 Sprüngli 或 Läderach 高級巧克力的半價甚至 1/4 價。

稍微有心一點的話買 Migros 的 Frey 牌或 Coop 的 Halba 牌巧克力，稍微貴一點點，也很適合送人。另外省瑞小秘訣就是超市 Denner 常常會有超大包巧克力大特價（可以看名字叫做：Neapolitans），裡面會是一小塊一小塊的通常是牛奶巧克力，也很適合買回去打發不太熟的同學或同事。

如果真的忘記買的話，蘇黎世機場還有超大 Coop 和 Migros 超市讓你最後採購，都在安檢前，可以買完塞進行李箱，拿去托運。也有 Läderach 和蘇黎世老牌 Sprüngli 巧克力，安檢前後都有，也可最後採購。詳情可看：{% assign target_post = site.posts | where: "url", "/blog/zurich-airport-guide/" | first %}<a href="{{ target_post.url }}" class="swiss-red-link">{{ target_post.title }}</a>
